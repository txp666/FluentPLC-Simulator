cmake_minimum_required(VERSION 3.21)
project(FluentPLCSimulator VERSION 0.1.1 LANGUAGES CXX)
file(WRITE "${CMAKE_BINARY_DIR}/FluentPLCSimulator-version.txt" "${PROJECT_VERSION}\n")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(FPLC_BUILD_TESTS "Build simulator protocol tests" ON)

set(FLUENTQTWIDGETS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/FluentQtWidgets" CACHE PATH "FluentQtWidgets sources")
set(SNAP7_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/snap7" CACHE PATH "Official Snap7 sources")
if(NOT EXISTS "${SNAP7_ROOT}/src/lib/snap7_libmain.cpp" OR NOT EXISTS "${FLUENTQTWIDGETS_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Initialize dependencies: git submodule update --init --recursive")
endif()
include("${FLUENTQTWIDGETS_ROOT}/cmake/FqwLocateQt.cmake")
fqw_locate_qt(6.12)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Svg Network)
find_package(Threads REQUIRED)
add_executable(render_logo tools/render_logo.cpp)
set_target_properties(render_logo PROPERTIES AUTOMOC OFF)
target_link_libraries(render_logo PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)

# Keep the pinned submodule untouched; apply modern macOS fixes to a build copy.
set(SNAP7_SOURCE_ROOT "${SNAP7_ROOT}")
if(APPLE)
    set(SNAP7_SOURCE_ROOT "${CMAKE_BINARY_DIR}/_deps/snap7-src")
    file(GLOB_RECURSE SNAP7_ORIGINAL_SOURCES RELATIVE "${SNAP7_ROOT}"
        "${SNAP7_ROOT}/src/*.h" "${SNAP7_ROOT}/src/*.cpp")
    foreach(source IN LISTS SNAP7_ORIGINAL_SOURCES)
        configure_file("${SNAP7_ROOT}/${source}" "${SNAP7_SOURCE_ROOT}/${source}" COPYONLY)
    endforeach()
    execute_process(COMMAND /usr/bin/patch --batch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/patches/snap7-macos.patch"
        WORKING_DIRECTORY "${SNAP7_SOURCE_ROOT}" RESULT_VARIABLE patch_result OUTPUT_QUIET ERROR_VARIABLE patch_error)
    if(NOT patch_result EQUAL 0)
        message(FATAL_ERROR "Cannot apply Snap7 macOS compatibility patch: ${patch_error}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/patches/snap7-macos.patch")
endif()

# The original OSX makefile links librt, which isn't available on modern macOS.
set(SNAP7_SOURCES
    src/sys/snap_msgsock.cpp src/sys/snap_sysutils.cpp src/sys/snap_tcpsrvr.cpp src/sys/snap_threads.cpp
    src/core/s7_client.cpp src/core/s7_isotcp.cpp src/core/s7_partner.cpp src/core/s7_peer.cpp
    src/core/s7_server.cpp src/core/s7_text.cpp src/core/s7_micro_client.cpp src/lib/snap7_libmain.cpp)
list(TRANSFORM SNAP7_SOURCES PREPEND "${SNAP7_SOURCE_ROOT}/")
add_library(snap7 SHARED ${SNAP7_SOURCES})
set_target_properties(snap7 PROPERTIES AUTOMOC OFF VERSION 1.4.3 SOVERSION 1
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(snap7 PRIVATE "${SNAP7_SOURCE_ROOT}/src/sys" "${SNAP7_SOURCE_ROOT}/src/core" "${SNAP7_SOURCE_ROOT}/src/lib")
target_link_libraries(snap7 PRIVATE Threads::Threads)
if(WIN32)
    target_compile_definitions(snap7 PRIVATE _CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS WIN32_LEAN_AND_MEAN)
    target_link_libraries(snap7 PRIVATE ws2_32 winmm)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(snap7 PRIVATE rt)
endif()
add_library(snap7_cpp STATIC "${SNAP7_ROOT}/release/wrappers/c-cpp/snap7.cpp")
set_target_properties(snap7_cpp PROPERTIES AUTOMOC OFF)
target_include_directories(snap7_cpp PUBLIC "${SNAP7_ROOT}/release/wrappers/c-cpp")
target_link_libraries(snap7_cpp PUBLIC snap7)

set(CMAKE_DISABLE_FIND_PACKAGE_Qt6Multimedia ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6MultimediaWidgets ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6WebEngineWidgets ON CACHE BOOL "" FORCE)
set(FQW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FQW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FQW_BUILD_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory("${FLUENTQTWIDGETS_ROOT}" "${CMAKE_BINARY_DIR}/_deps/fluentqtwidgets" EXCLUDE_FROM_ALL)

add_library(fplc_core STATIC src/core/project.h src/core/project.cpp src/core/address.h src/core/address.cpp
    src/core/codec.h src/core/codec.cpp src/core/i18n.h src/core/i18n.cpp resources/i18n.qrc
    src/plc/plc_server.h src/plc/plc_server.cpp)
target_include_directories(fplc_core PUBLIC src)
target_link_libraries(fplc_core PUBLIC Qt6::Core Qt6::Network snap7_cpp)

add_executable(FluentPLCServer src/server_main.cpp)
target_link_libraries(FluentPLCServer PRIVATE fplc_core)
add_library(fplc_update STATIC src/update/update_release.h src/update/update_release.cpp
    src/update/update_service.h src/update/update_service.cpp
    src/update/update_installer.h src/update/update_installer.cpp)
target_include_directories(fplc_update PUBLIC src)
target_link_libraries(fplc_update PUBLIC Qt6::Core Qt6::Network)
target_compile_definitions(fplc_update PUBLIC FPLC_VERSION="${PROJECT_VERSION}")
add_library(fplc_ui STATIC src/app/main_window.h src/app/main_window.cpp
    src/update/update_controller.h src/update/update_controller.cpp
    src/update/update_dialog.h src/update/update_dialog.cpp)
target_link_libraries(fplc_ui PUBLIC fplc_core fplc_update FluentQtWidgets::Widgets Qt6::Widgets Qt6::Svg)
add_executable(FluentPLCSimulator WIN32 MACOSX_BUNDLE src/main.cpp resources/app.qrc)
target_link_libraries(FluentPLCSimulator PRIVATE fplc_ui)
target_compile_definitions(FluentPLCSimulator PRIVATE FPLC_VERSION="${PROJECT_VERSION}")
set_target_properties(FluentPLCSimulator PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "FluentPLC Simulator"
    MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.txp666.FluentPLCSimulator"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
if(APPLE)
    target_sources(FluentPLCSimulator PRIVATE resources/logo.icns)
    set_source_files_properties(resources/logo.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(FluentPLCSimulator PROPERTIES MACOSX_BUNDLE_ICON_FILE logo.icns)
elseif(WIN32)
    enable_language(RC)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/fplc_icon.rc"
        CONTENT "1 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/resources/logo.ico\"\n")
    target_sources(FluentPLCSimulator PRIVATE "${CMAKE_BINARY_DIR}/fplc_icon.rc")
endif()
if(MSVC)
    target_compile_options(fplc_core PRIVATE /utf-8)
    target_compile_options(fplc_ui PRIVATE /utf-8)
    target_compile_options(fplc_update PRIVATE /utf-8)
    target_compile_options(FluentPLCServer PRIVATE /utf-8)
    target_compile_options(FluentPLCSimulator PRIVATE /utf-8)
endif()
if(WIN32)
    foreach(app FluentPLCSimulator FluentPLCServer)
        add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:snap7> $<TARGET_FILE_DIR:${app}>)
